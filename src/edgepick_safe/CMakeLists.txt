cmake_minimum_required(VERSION 3.22)
project(edgepick_safe)
find_package(ament_cmake REQUIRED)
find_package(ament_cmake_python REQUIRED)
foreach(dep rclcpp edgepick_interfaces sensor_msgs geometry_msgs std_msgs std_srvs moveit_ros_planning_interface moveit_ros_planning tf2_ros tf2_geometry_msgs moveit_msgs geometric_shapes)
  find_package(${dep} REQUIRED)
endforeach()
add_executable(safety_executor src/safety_executor.cpp)
target_compile_features(safety_executor PUBLIC cxx_std_17)
target_compile_options(safety_executor PRIVATE -Wall -Wextra -Wpedantic)
target_include_directories(safety_executor PRIVATE include)
ament_target_dependencies(safety_executor rclcpp edgepick_interfaces sensor_msgs geometry_msgs std_msgs std_srvs moveit_ros_planning_interface moveit_ros_planning tf2_ros tf2_geometry_msgs moveit_msgs geometric_shapes)
ament_python_install_package(${PROJECT_NAME})
install(TARGETS safety_executor DESTINATION lib/${PROJECT_NAME})
install(PROGRAMS scripts/policy_node scripts/perception_node scripts/mock_inputs scripts/mujoco_backend DESTINATION lib/${PROJECT_NAME})
install(DIRECTORY launch config DESTINATION share/${PROJECT_NAME})
if(BUILD_TESTING)
  find_package(ament_cmake_gtest REQUIRED)
  find_package(ament_cmake_pytest REQUIRED)
  ament_add_gtest(gate_test test/gate_test.cpp)
  target_include_directories(gate_test PRIVATE include)
  target_compile_features(gate_test PUBLIC cxx_std_17)
  ament_add_pytest_test(contract_test test/test_contract.py)
  ament_add_pytest_test(perception_test test/test_perception.py)
endif()
ament_package()
